cmake_minimum_required(VERSION 3.16)
project(arli_external_consumer LANGUAGES CXX)
option(TEST_EMBED "Also consume Qt embedding SDK" OFF)
if(TEST_EMBED)
  find_package(Arli CONFIG REQUIRED COMPONENTS client embed)
else()
  find_package(Arli CONFIG REQUIRED COMPONENTS client)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE Arli::client Arli::grpc_tools)
if(TEST_EMBED)
  target_link_libraries(consumer PRIVATE Arli::embed)
  target_compile_definitions(consumer PRIVATE TEST_EMBED)
endif()

if(NOT TEST_EMBED)
  add_executable(daemon-roundtrip daemon_roundtrip.cpp)
  target_link_libraries(daemon-roundtrip PRIVATE Arli::client)
endif()
